from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent
SOURCE_DIR = Path(__file__).resolve().parent
OUTPUT_FILE = PROJECT_ROOT / "ai_code_dump.txt"

EXCLUDED_DIRS = {
    "__pycache__",
    ".venv",
    "venv",
    "node_modules",
}

INCLUDED_EXTENSIONS = {
    ".py",
    ".txt",
    ".md",
    ".json",
    ".yaml",
    ".yml",
}

def dump_directory():
    if not SOURCE_DIR.exists():
        print(f"Directory not found: {SOURCE_DIR}")
        return

    files = sorted(
        path for path in SOURCE_DIR.rglob("*")
        if path.is_file()
        and path.suffix.lower() in INCLUDED_EXTENSIONS
        and not any(part in EXCLUDED_DIRS for part in path.parts)
    )

    with OUTPUT_FILE.open("w", encoding="utf-8") as output:
        output.write("AI SERVICES CODE DUMP\n")
        output.write("=" * 80 + "\n")
        output.write(f"Source: {SOURCE_DIR}\n")
        output.write(f"Total files: {len(files)}\n")
        output.write("=" * 80 + "\n\n")

        for file in files:
            relative_path = file.relative_to(PROJECT_ROOT)

            output.write("\n\n")
            output.write("=" * 80 + "\n")
            output.write(f"FILE: {relative_path}\n")
            output.write("=" * 80 + "\n\n")

            try:
                content = file.read_text(encoding="utf-8")
                output.write(content)
                output.write("\n")
            except (UnicodeDecodeError, OSError) as error:
                output.write(f"[Could not read file: {error}]\n")

    print(f"Dump completed!")
    print(f"Files included: {len(files)}")
    print(f"Output: {OUTPUT_FILE}")


if __name__ == "__main__":
    dump_directory()

class EpisodeManager:

    def generate_episode_id(self):

        return f"EP_{uuid4().hex[:8].upper()}"


    def get_active_episode(self, db, user_id):

        return db.query(Episode).filter(

            Episode.user_id == user_id,

            Episode.status == "ACTIVE"

        ).first()


    def create_episode(self, db, user_id, title):

        episode = Episode(

            episode_id=self.generate_episode_id(),

            user_id=user_id,

            title=title,

            status="ACTIVE"

        )

        db.add(episode)

        db.commit()

        db.refresh(episode)

        return episode


    def update_summary(self, db, episode, summary):

        episode.summary = summary

        db.commit()

        db.refresh(episode)

        return episode


    def close_episode(self, db, episode):

        episode.status = "CLOSED"

        episode.closed_at = datetime.utcnow()

        db.commit()

        db.refresh(episode)

        return episode

import chromadb

from chromadb.utils.embedding_functions import (
    SentenceTransformerEmbeddingFunction
)


class KnowledgeRAG:

    def __init__(self):

        embedding_function = SentenceTransformerEmbeddingFunction(

            model_name="BAAI/bge-small-en-v1.5"

        )

        client = chromadb.PersistentClient(

            path="vector_db"

        )

        self.collection = client.get_collection(

            name="knowledge",

            embedding_function=embedding_function

        )

        ##################################################
        # Similarity Threshold
        ##################################################

        self.threshold = 0.60

    def retrieve(

        self,

        query

    ):

        ##################################################
        # Search
        ##################################################

        results = self.collection.query(

            query_texts=[query],

            n_results=3,

            include=[

                "documents",

                "distances"

            ]

        )

        documents = results["documents"][0]

        distances = results["distances"][0]

        ##################################################
        # Nothing Retrieved
        ##################################################

        if len(documents) == 0:

            return None

        ##################################################
        # Best Match Too Far
        ##################################################

        if distances[0] > self.threshold:

            print("\n===== Knowledge RAG =====")
            print("No relevant knowledge found.")
            print(f"Distance: {distances[0]:.3f}")
            print("=========================\n")

            return None

        ##################################################
        # Build Knowledge
        ##################################################

        knowledge = ""

        for document, distance in zip(

            documents,

            distances

        ):

            if distance <= self.threshold:

                knowledge += document

                knowledge += "\n\n"

        ##################################################
        # Debug
        ##################################################

        print("\n===== Knowledge Retrieved =====")

        print(

            knowledge

        )

        print("Distances:", distances)

        print("===============================\n")

        return knowledge.strip()

import json
import logging

from sqlalchemy.orm import Session

from models import (
    EpisodeReflection,
    LongTermMemory as LongTermMemoryModel
)

from services.ai.memory_consolidator import MemoryConsolidator
from services.ai.profile import Profile


logger = logging.getLogger(__name__)


class LongTermMemory:

    def __init__(self):
        self.memory_consolidator = MemoryConsolidator()
        self.profile_generator = Profile()

    # =====================================================
    # UPDATE LONG-TERM MEMORY AND PROFILE
    # =====================================================

    def update(
        self,
        db: Session,
        user_id,
        episode
    ):

        if episode is None:
            raise ValueError("Episode does not exist.")

        if episode.user_id != user_id:
            raise ValueError(
                "Episode does not belong to this user."
            )

        if not episode.summary or not episode.summary.strip():
            raise ValueError("Episode summary is empty.")

        try:

            print("\n========== LONG-TERM MEMORY UPDATE ==========")
            print("User ID:", user_id)
            print("Episode ID:", episode.episode_id)

            # -------------------------------------------------
            # 1. LOAD EXISTING MEMORY RECORD
            # -------------------------------------------------

            record = (
                db.query(LongTermMemoryModel)
                .filter(
                    LongTermMemoryModel.user_id == user_id
                )
                .with_for_update()
                .one_or_none()
            )

            previous_memory = (
                record.memory
                if record and record.memory
                else None
            )

            previous_themes = (
                record.profile_themes
                if record and isinstance(record.profile_themes, list)
                else []
            )

            print("\nPrevious Long-Term Memory:")
            print(previous_memory or "No previous memory.")

            print("\nPrevious Profile Themes:")
            print(previous_themes)

            # -------------------------------------------------
            # 2. LOAD EPISODE REFLECTIONS
            # -------------------------------------------------

            reflections = (
                db.query(EpisodeReflection)
                .filter(
                    EpisodeReflection.episode_id
                    == episode.episode_id
                )
                .order_by(
                    EpisodeReflection.created_at
                )
                .all()
            )

            print(
                "\nEpisode reflections:",
                len(reflections)
            )

            # -------------------------------------------------
            # 3. CONSOLIDATE LONG-TERM MEMORY
            # -------------------------------------------------

            updated_memory = self.memory_consolidator.consolidate(
                previous_memory=previous_memory,
                summary=episode.summary,
                reflections=reflections
            )

            if (
                not isinstance(updated_memory, str)
                or not updated_memory.strip()
            ):
                raise ValueError(
                    "Memory consolidation returned empty memory."
                )

            updated_memory = updated_memory.strip()

            print("\nUpdated Long-Term Memory:")
            print(updated_memory)

            # -------------------------------------------------
            # 4. GENERATE PROFILE THEMES
            # -------------------------------------------------

            # IMPORTANT:
            # Profile generation must never prevent memory saving.
            # If the LLM fails, retain the previous themes.

            updated_themes = previous_themes

            print("\nGenerating Profile Themes...")

            try:

                generated_themes = self.profile_generator.generate(
                    memory=updated_memory,
                    previous_themes=previous_themes
                )

                if not isinstance(generated_themes, list):
                    raise ValueError(
                        "Profile generator did not return a list."
                    )

                # Only accept dictionary-based themes.
                valid_themes = [
                    theme
                    for theme in generated_themes
                    if isinstance(theme, dict)
                ]

                if generated_themes and not valid_themes:
                    raise ValueError(
                        "Profile generator returned no valid theme objects."
                    )

                # An empty list is allowed when there are no meaningful
                # themes, but do not erase existing themes accidentally.
                if valid_themes:
                    # Verify that the result can be stored in a JSON column.
                    json.dumps(valid_themes)

                    updated_themes = valid_themes

                elif previous_themes:
                    logger.warning(
                        "Profile generator returned no themes. "
                        "Preserving previous themes for user %s.",
                        user_id
                    )

                print("\nProfile Themes:")
                print(updated_themes)

            except Exception:

                logger.exception(
                    "Profile generation failed for user %s. "
                    "Preserving previous themes.",
                    user_id
                )

                updated_themes = previous_themes

            # -------------------------------------------------
            # 5. SAVE MEMORY AND THEMES TOGETHER
            # -------------------------------------------------

            if record is None:

                record = LongTermMemoryModel(
                    user_id=user_id,
                    memory=updated_memory,
                    profile_themes=updated_themes
                )

                db.add(record)

            else:

                record.memory = updated_memory
                record.profile_themes = updated_themes

            # Commit consolidated memory even if Profile generation failed.
            db.commit()
            db.refresh(record)

            print("\n========== MEMORY UPDATE SUCCESS ==========")
            print("Memory saved:", bool(record.memory))
            print(
                "Profile themes saved:",
                len(record.profile_themes or [])
            )
            print("Updated at:", record.updated_at)
            print("===========================================\n")

            return record.memory

        except Exception:

            db.rollback()

            logger.exception(
                "Long-term memory update failed "
                "for user %s, episode %s",
                user_id,
                getattr(episode, "episode_id", None)
            )

            raise

    # =====================================================
    # RETRIEVE LONG-TERM MEMORY
    # =====================================================

    def retrieve(
        self,
        db: Session,
        user_id
    ):

        record = (
            db.query(LongTermMemoryModel)
            .filter(
                LongTermMemoryModel.user_id == user_id
            )
            .one_or_none()
        )

        if record is None:
            return None

        return record.memory

    # =====================================================
    # RETRIEVE PROFILE THEMES
    # =====================================================

    def retrieve_profile(
        self,
        db: Session,
        user_id
    ):

        record = (
            db.query(LongTermMemoryModel)
            .filter(
                LongTermMemoryModel.user_id == user_id
            )
            .one_or_none()
        )

        if record is None:
            return []

        return record.profile_themes or []

import logging

from services.llm_provider import generate_response
from services.ai.prompts.memory_consolidator_prompt import PROMPT
from models import LongTermMemory
from sqlalchemy.orm import Session

logger = logging.getLogger(__name__)


class MemoryConsolidator:

    def consolidate(self, previous_memory, summary, reflections):

        if not summary or not summary.strip():
            raise ValueError("Episode summary is empty.")

        prompt = self.build_prompt(
            previous_memory,
            summary,
            reflections
        )

        response = generate_response(prompt)

        if not isinstance(response, str) or not response.strip():
            raise ValueError("LLM returned an empty memory.")

        memory = response.strip()

        print("\n===== Generated Long-Term Memory =====")
        print(memory)
        print("=======================================\n")

        return memory

    def build_prompt(self, previous_memory, summary, reflections):

        previous_memory = (
            previous_memory.strip()
            if previous_memory
            else "No previous memory exists."
        )

        reflections = reflections or []

        reflections_text = "\n".join(
            f"- {r.reflection}"
            for r in reflections
            if r.reflection
        )

        if not reflections_text:
            reflections_text = "No reflections available."

        return PROMPT.format(
            memory=previous_memory,
            summary=summary.strip(),
            reflections=reflections_text
        )

    def load(self, db: Session, user_id):

        record = db.query(LongTermMemory).filter(
            LongTermMemory.user_id == user_id
        ).first()

        return record.memory if record else None

    def save(self, db: Session, user_id, memory):

        if not memory or not memory.strip():
            raise ValueError("Refusing to save empty long-term memory.")

        try:
            existing = db.query(LongTermMemory).filter(
                LongTermMemory.user_id == user_id
            ).first()

            if existing:
                existing.memory = memory.strip()
            else:
                existing = LongTermMemory(
                    user_id=user_id,
                    memory=memory.strip()
                )
                db.add(existing)

            db.commit()
            db.refresh(existing)

            print("\n===== Long-Term Memory Saved =====")
            print(existing.memory)
            print("==================================\n")

            return existing.memory

        except Exception:
            db.rollback()
            logger.exception("Failed to save long-term memory.")
            raise

import logging

from services.llm_provider import generate_response
from services.ai.json_parser import JSONParser
from services.ai.prompts.mood_prompt import PROMPT

from models import EpisodeMood


logger = logging.getLogger(__name__)


# IMPORTANT:
# These labels must match the moods supported by your application
# and your existing mood database conventions.
MOOD_LABELS = {
    "happy",
    "sad",
    "angry",
    "fear",
    "disgust",
    "surprise",
    "neutral",
}

MIN_CONFIDENCE = 0.75
MAX_CONTEXT_MESSAGES = 6
MAX_TIMELINE_ENTRIES = 5


class MoodDetector:

    def __init__(self):
        self.json_parser = JSONParser()

    def update(
        self,
        db,
        episode,
        conversation,
        latest_message,
        reply,
    ):
        """
        Call once for each new user turn, usually after the assistant
        has generated its reply.

        The detector predicts the user's mood and saves it only if:
        - The result is valid.
        - Confidence meets the threshold.
        - Supporting evidence is provided.
        - The mood differs from the latest saved mood.
        """

        if not latest_message or not latest_message.strip():
            return

        try:
            # Load the saved mood timeline.
            previous_moods = (
                db.query(EpisodeMood)
                .filter(
                    EpisodeMood.episode_id == episode.episode_id
                )
                .order_by(EpisodeMood.created_at.asc())
                .all()
            )

            # Only pass a small amount of recent dialogue to the LLM.
            recent_context = self._get_recent_context(
                conversation=conversation,
                latest_message=latest_message,
                reply=reply,
            )

            # Avoid putting a long mood history into every prompt.
            recent_moods = previous_moods[-MAX_TIMELINE_ENTRIES:]

            prompt = self.build_prompt(
                episode=episode,
                previous_moods=recent_moods,
                conversation=recent_context,
                latest_message=latest_message,
                reply=reply,
            )

            response = generate_response(prompt)

            if not response:
                return

            decision = self.json_parser.parse(response)

            # Reject malformed or unexpected output.
            if not isinstance(decision, dict):
                logger.debug("Mood detection returned invalid JSON.")
                return

            mood = self._normalise_mood(decision.get("mood"))

            # The model may explicitly say it cannot determine a mood.
            if mood is None:
                return

            confidence = self._normalise_confidence(
                decision.get("confidence")
            )

            # Missing or invalid confidence is not accepted.
            if confidence is None:
                return

            if confidence < MIN_CONFIDENCE:
                return

            evidence = decision.get("evidence")

            if not isinstance(evidence, str) or not evidence.strip():
                return

            # Do not add another database entry for the current mood.
            if previous_moods:
                latest_saved_mood = self._normalise_mood(
                    previous_moods[-1].mood
                )

                if latest_saved_mood == mood:
                    return

            # A genuine, sufficiently confident mood change.
            self.save(
                db=db,
                episode=episode,
                mood=mood,
            )

        except Exception:
            # Avoid printing prompts and full conversation contents,
            # which can expose sensitive information in terminal logs.
            db.rollback()
            logger.exception(
                "Mood detection failed for episode %s",
                getattr(episode, "episode_id", "unknown"),
            )

    def _get_recent_context(
        self,
        conversation,
        latest_message,
        reply,
    ):
        """
        Return prior dialogue only. The latest user message and reply
        are supplied separately in the prompt.

        This avoids accidentally presenting the current turn twice.
        """

        messages = []

        for item in conversation or []:
            if not isinstance(item, dict):
                continue

            role = str(item.get("role", "")).strip().lower()
            content = item.get("content", "")

            if role not in {"user", "assistant"}:
                continue

            if not isinstance(content, str) or not content.strip():
                continue

            messages.append({
                "role": role,
                "content": content.strip(),
            })

        # Remove the reply if the caller already included it.
        if (
            messages
            and messages[-1]["role"] == "assistant"
            and messages[-1]["content"] == (reply or "").strip()
        ):
            messages.pop()

        # Remove the latest user turn if the caller already included it.
        if (
            messages
            and messages[-1]["role"] == "user"
            and messages[-1]["content"] == latest_message.strip()
        ):
            messages.pop()

        return messages[-MAX_CONTEXT_MESSAGES:]

    def _normalise_mood(self, value):
        """Return a canonical mood label or None."""

        if not isinstance(value, str):
            return None

        mood = value.strip().lower()

        if mood in {"", "none", "null", "unknown", "unclear"}:
            return None

        if mood not in MOOD_LABELS:
            return None

        return mood

    def _normalise_confidence(self, value):
        """Accept numeric confidence values between 0 and 1."""

        if isinstance(value, bool):
            return None

        try:
            confidence = float(value)
        except (TypeError, ValueError):
            return None

        if not 0.0 <= confidence <= 1.0:
            return None

        return confidence

    def build_prompt(
        self,
        episode,
        previous_moods,
        conversation,
        latest_message,
        reply,
    ):
        episode_text = f"""
Episode ID: {episode.episode_id}
Title: {episode.title or "Untitled"}
Status: {episode.status}
Summary: {episode.summary or "None"}
"""

        if not previous_moods:
            moods_text = "None recorded."
        else:
            moods_text = "\n".join(
                f"{m.created_at} - {m.mood}"
                for m in previous_moods
            )

        if not conversation:
            conversation_text = "No prior messages available."
        else:
            conversation_text = "\n".join(
                f"{m['role'].capitalize()}: {m['content']}"
                for m in conversation
            )

        return PROMPT.format(
            allowed_moods=", ".join(sorted(MOOD_LABELS)),
            episode=episode_text,
            moods=moods_text,
            conversation=conversation_text,
            message=latest_message,
            reply=reply or "No assistant reply available.",
        )

    def save(self, db, episode, mood):
        """Persist one mood transition."""

        new_mood = EpisodeMood(
            episode_id=episode.episode_id,
            mood=mood,
        )

        try:
            db.add(new_mood)
            db.commit()
            db.refresh(new_mood)

            logger.debug(
                "Saved mood transition: episode=%s, mood=%s",
                episode.episode_id,
                mood,
            )

        except Exception:
            db.rollback()
            raise

    # =====================================================
    # EPISODE LIFECYCLE
    # =====================================================

    def _handle_episode_lifecycle(
        self,
        db,
        user_id,
        conversation,
        latest_message,
        episode,
        decision
    ):

        long_term_memory_updated = False
        closed_episode_id = None

        action = decision.get("action")

                print("Episode:", episode.episode_id)

                # 1. Generate episode summary.
                summary = self.episode_summarizer.summarize(
                    conversation
                )

                if not summary or not summary.strip():
                    raise ValueError(
                        "Episode summarization returned an empty summary."
                    )

                # 2. Save episode summary.
                self.episode_summarizer.save(
                    db,
                    episode,
                    summary
                )

                # 3. Update cumulative long-term memory.
                # This method loads previous memory, retrieves
                # reflections, consolidates and saves the result.

                updated_memory = self.long_term_memory.update(
                    db,
                    user_id,
                    episode
                )

                if not updated_memory or not updated_memory.strip():
                    raise ValueError(
                        "Long-term memory update failed."
                    )

                print("\nLong-term memory successfully updated.")

                # 4. Preserve the closed episode ID.
                closed_episode_id = episode.episode_id

                # 5. Close episode only after memory is saved.
                self.episode_manager.close_episode(
                    db,
                    episode
                )

                # 6. Notify the frontend through the response.
                long_term_memory_updated = True

                print("Episode closed:", closed_episode_id)
                print("Long-term memory updated:", True)
                print("====================================\n")

            # No active episode remains after CLOSE.
            episode = None

        elif action == "CONTINUE":

            pass

        elif action == "NONE":

            pass

        return (
            episode,
            long_term_memory_updated,
            closed_episode_id
        )

    # =====================================================
    # NON-STREAMING CHAT
    # =====================================================

    def chat(
        self,
        db,
        user_id,
        conversation,
        latest_message,
        current_title
    ):

        # 1. Generate conversation title.
        title = None

        if current_title == "New Conversation":

            title = generate_title(
                latest_message
            )

        # 2. Get active episode.
        episode = self.episode_manager.get_active_episode(
            db,
            user_id
        )

        # 3. Detect episode action.
        decision = self.episode_detector.detect(
            episode,
            latest_message
        )

        # 4. Handle episode lifecycle.
        (
            episode,
            long_term_memory_updated,
            closed_episode_id
        ) = self._handle_episode_lifecycle(
            db=db,
            user_id=user_id,
            conversation=conversation,
            latest_message=latest_message,
            episode=episode,
            decision=decision
        )

        # 5. Retrieve current long-term memory.
        # This retrieves the newly updated memory when CLOSE occurred.
        long_term_memory_text = self.long_term_memory.retrieve(
            db,
            user_id
        )

        # 6. Retrieve knowledge.
        knowledge = self.knowledge_rag.retrieve(
            latest_message
        )

        # 7. Generate assistant response.
        reply = chat(
            conversation=conversation,
            profile=None,
            memory=long_term_memory_text,
            knowledge=knowledge,
            episode=episode,
            reflection=None,
            mood=None
        )

        # 8. Mood detection.
        if episode is not None:

            self.mood_detector.update(
                db,
                episode,
                conversation,
                latest_message,
                reply
            )

        # 9. Reflection generation.
        if episode is not None:

            self.reflection_generator.update(
                db,
                episode,
                conversation,
                latest_message,
                reply
            )

        # 10. Return response.
        return {

            "reply": reply,

            "title": title,

            "episode": episode,

            "episode_id": (
                episode.episode_id
                if episode is not None
                else None
            ),

            "closed_episode_id": closed_episode_id,

            "long_term_memory_updated": long_term_memory_updated

        }

    # =====================================================
    # STREAMING CHAT
    # =====================================================

    def chat_stream(
        self,
        db,
        user_id,
        conversation,
        latest_message,
        current_title
    ):

        # 1. Generate conversation title.
        title = None

        if current_title == "New Conversation":

            title = generate_title(
                latest_message
            )

        # 2. Get active episode.
        episode = self.episode_manager.get_active_episode(
            db,
            user_id
        )

        # 3. Detect episode action.
        decision = self.episode_detector.detect(
            episode,
            latest_message
        )

        # 4. Handle episode lifecycle.
        (
            episode,
            long_term_memory_updated,
            closed_episode_id
        ) = self._handle_episode_lifecycle(
            db=db,
            user_id=user_id,
            conversation=conversation,
            latest_message=latest_message,
            episode=episode,
            decision=decision
        )

        # 5. Retrieve latest long-term memory.
        long_term_memory_text = self.long_term_memory.retrieve(
            db,
            user_id
        )

        # 6. Retrieve knowledge.
        knowledge = self.knowledge_rag.retrieve(
            latest_message
        )

        # 7. Stream assistant response.
        reply_parts = []

        for chunk in llm_chat_stream(
            conversation=conversation,
            profile=None,
            memory=long_term_memory_text,
            knowledge=knowledge,
            episode=episode,
            reflection=None,
            mood=None
        ):

            reply_parts.append(chunk)

            yield {

                "type": "chunk",

                "content": chunk

            }

        # 8. Combine streamed response.
        reply = "".join(reply_parts)

        # 9. Mood detection.
        if episode is not None:

            self.mood_detector.update(
                db,
                episode,
                conversation,
                latest_message,
                reply
            )

        # 10. Reflection generation.
        if episode is not None:

            self.reflection_generator.update(
                db,
                episode,
                conversation,
                latest_message,
                reply
            )

        # 11. Send completion event.
        # The frontend can now refresh the Profile.
        yield {

            "type": "done",

            "title": title,

            "episode_id": (
                episode.episode_id
                if episode is not None
                else None
            ),

            "closed_episode_id": closed_episode_id,

            "long_term_memory_updated": long_term_memory_updated

        }

import json
import hashlib
import logging
import re

from services.llm_provider import generate_response
from services.ai.prompts.profile_generator_prompt import PROMPT

logger = logging.getLogger(__name__)


class Profile:

    MAX_THEMES = 8

    ALLOWED_TONES = {
        "amber",
        "green",
        "blue",
        "purple"
    }

    DEFAULT_TONE = "blue"

    # =====================================================
    # BUILD PROMPT
    # =====================================================

    def build_prompt(self, memory, previous_themes=None):
        previous_themes_json = json.dumps(
            previous_themes or [],
            ensure_ascii=False,
            indent=2
        )

        return (
            PROMPT
            .replace("{memory}", str(memory))
            .replace("{previous_themes}", previous_themes_json)
        )

        return self.PROMPT.format(
            memory=memory.strip(),
            previous_themes=previous_themes_json
        )

    # =====================================================
    # GENERATE PROFILE
    # =====================================================

    def generate(self, memory, previous_themes=None):

        """
        Generate Profile themes from complete long-term memory.

        Args:
            memory:
                Complete consolidated long-term memory.

            previous_themes:
                Previously saved Profile themes.

        Returns:
            List of validated Profile theme dictionaries.
        """

        if not memory or not memory.strip():
            logger.warning(
                "Cannot generate Profile from empty memory."
            )

            return []

        prompt = self.build_prompt(
            memory,
            previous_themes
        )

        logger.info(
            "Generating Profile themes from long-term memory."
        )

        response = generate_response(prompt, max_completion_tokens=4096)

        if not isinstance(response, str) or not response.strip():
            raise ValueError(
                "Profile generator returned an empty response."
            )

        # Parse LLM response.
        parsed = self.parse_response(response)

        # Validate and normalize themes.
        themes = self.validate_themes(parsed)

        if not themes:
            raise ValueError(
                "Profile generator returned no valid themes."
            )

        logger.info(
            "Successfully generated %d Profile themes.",
            len(themes)
        )

        print("\n========== GENERATED PROFILE ==========")

        print(
            json.dumps(
                themes,
                indent=2,
                ensure_ascii=False
            )
        )

        print("=======================================\n")

        return themes

    # =====================================================
    # PARSE JSON RESPONSE
    # =====================================================

    def parse_response(self, response):

        """
        Parse JSON even when the LLM accidentally wraps
        the response inside markdown code fences or text.
        """

        cleaned = response.strip()

        # Remove markdown code fences.
        cleaned = re.sub(
            r"^```(?:json)?\s*",
            "",
            cleaned,
            flags=re.IGNORECASE
        )

        cleaned = re.sub(
            r"\s*```$",
            "",
            cleaned
        )

        # Find the beginning of the JSON.
        decoder = json.JSONDecoder()

        start_positions = [
            position
            for position in (
                cleaned.find("{"),
                cleaned.find("[")
            )
            if position >= 0
        ]

        if not start_positions:
            raise ValueError(
                "No JSON object or array found in Profile response."
            )

        start = min(start_positions)

        try:

            parsed, _ = decoder.raw_decode(
                cleaned[start:]
            )

            return parsed

        except json.JSONDecodeError as exc:

            logger.exception(
                "Could not parse Profile generator JSON."
            )

            raise ValueError(
                "Invalid JSON returned by Profile generator."
            ) from exc

    # =====================================================
    # VALIDATE AND NORMALIZE THEMES
    # =====================================================

    def validate_themes(self, parsed):

        """
        Validate the theme structure and normalize each theme.

        Generates stable IDs based on category and title.
        """

        if isinstance(parsed, list):
            raw_themes = parsed

        elif isinstance(parsed, dict):
            raw_themes = parsed.get("themes", [])

        else:
            raise ValueError(
                "Invalid Profile JSON structure."
            )

        if not isinstance(raw_themes, list):
            raise ValueError(
                "Profile themes must be a list."
            )

        themes = []

        seen = set()

        for item in raw_themes:

            if not isinstance(item, dict):
                continue

            category = self.clean_text(
                item.get("category"),
                max_length=40
            )

            title = self.clean_text(
                item.get("title"),
                max_length=60
            )

            summary = self.clean_text(
                item.get("summary"),
                max_length=120
            )

            details = self.clean_text(
                item.get("details"),
                max_length=600
            )

            tone = self.clean_text(
                item.get("tone"),
                max_length=20
            ).lower()

            # Required fields.
            if not title or not details:
                continue

            if not category:
                category = "Life theme"

            if not summary:
                summary = self.clean_text(
                    details,
                    max_length=100
                )

            # Only allow known visual tones.
            if tone not in self.ALLOWED_TONES:
                tone = self.DEFAULT_TONE

            # Generate stable ID.
            theme_id = self.generate_id(
                category,
                title
            )

            # Avoid duplicate category/title pairs.
            duplicate_key = (
                category.lower(),
                title.lower()
            )

            if duplicate_key in seen:
                continue

            seen.add(duplicate_key)

            themes.append({
                "id": theme_id,
                "category": category,
                "title": title,
                "summary": summary,
                "details": details,
                "tone": tone
            })

            # Keep Profile manageable.
            if len(themes) >= self.MAX_THEMES:
                break

        return themes

    # =====================================================
    # GENERATE STABLE THEME ID
    # =====================================================

    def generate_id(self, category, title):

        """
        Same category and title produce the same ID.

        This allows React to recognize existing themes
        when new themes are added.
        """

        value = (
            category.strip().lower()
            + "::"
            + title.strip().lower()
        )

        digest = hashlib.sha256(
            value.encode("utf-8")
        ).hexdigest()[:12]

        return f"theme_{digest}"

    # =====================================================
    # CLEAN TEXT
    # =====================================================

    def clean_text(self, value, max_length=500):

        if not isinstance(value, str):
            return ""

        return " ".join(
            value.split()
        )[:max_length].strip()

from services.llm_provider import generate_response
from services.ai.json_parser import JSONParser
from services.ai.prompts.reflection_prompt import PROMPT
from models import EpisodeReflection


class ReflectionGenerator:

    def __init__(self):
        self.json_parser = JSONParser()

    def update(
        self,
        db,
        episode,
        conversation,
        latest_message,
        reply
    ):
        print("\n" + "=" * 60)
        print("REFLECTION DEBUG: UPDATE STARTED")
        print("=" * 60)

        try:
            # 1. Validate episode
            if episode is None:
                print("[ERROR] No active episode. Cannot generate reflection.")
                return

            print(f"[INFO] Episode ID: {episode.episode_id}")
            print(f"[INFO] Episode title: {episode.title}")
            print(f"[INFO] Latest message: {latest_message!r}")
            print(f"[INFO] Conversation messages: {len(conversation)}")

            # 2. Increment counter
            episode.messages_since_last_reflection = (
                (episode.messages_since_last_reflection or 0) + 1
            )

            db.commit()
            db.refresh(episode)

            counter = episode.messages_since_last_reflection

            print(f"[DEBUG] Reflection counter: {counter}/2")

            # 3. Wait until threshold
            if counter < 1:
                print("[INFO] Waiting for another message.")
                return

            # 4. Load previous reflections
            previous_reflections = (
                db.query(EpisodeReflection)
                .filter(
                    EpisodeReflection.episode_id == episode.episode_id
                )
                .order_by(EpisodeReflection.created_at)
                .all()
            )

            print(
                f"[INFO] Previous reflections found: "
                f"{len(previous_reflections)}"
            )

            # 5. Build prompt
            prompt = self.build_prompt(
                episode,
                previous_reflections,
                conversation,
                latest_message,
                reply
            )

            print("\n[DEBUG] Reflection prompt:")
            print(prompt)

            # 6. Call LLM
            print("\n[INFO] Calling reflection LLM...")

            response = generate_response(prompt)

            print("\n[DEBUG] Raw LLM response:")
            print(repr(response))

            if not response or not response.strip():
                print("[ERROR] LLM returned an empty response.")
                return

            # 7. Parse JSON
            print("\n[INFO] Parsing LLM response...")

            decision = self.parse_reflection_response(response)

            print("[DEBUG] Parsed decision:", repr(decision))

            if not isinstance(decision, dict):
                print("[ERROR] Parsed response is not a JSON object.")
                return

            if "reflection" not in decision:
                print("[ERROR] Missing 'reflection' key in response.")
                return

            reflection = decision["reflection"]

            # 8. Handle no reflection
            if reflection is None:
                print(
                    "[INFO] LLM decided there is no meaningful "
                    "new insight."
                )
                self.reset_counter(db, episode)
                return

            if not isinstance(reflection, str) or not reflection.strip():
                print("[ERROR] Reflection is not a valid non-empty string.")
                return

            reflection = reflection.strip()

            # 9. Duplicate detection
            if previous_reflections:
                latest_reflection = previous_reflections[-1].reflection

                if (
                    latest_reflection
                    and latest_reflection.strip().casefold()
                    == reflection.casefold()
                ):
                    print("[INFO] Duplicate reflection detected; not saving.")
                    self.reset_counter(db, episode)
                    return

            # 10. Save reflection
            print("\n[INFO] Saving reflection to database...")

            self.save(db, episode, reflection)

            # Reset only after successful save
            self.reset_counter(db, episode)

            print("[SUCCESS] Reflection pipeline completed.")

        except Exception as exc:
            print("\n[ERROR] Reflection generation failed.")
            print(f"[ERROR] {type(exc).__name__}: {exc}")
            traceback.print_exc()

            try:
                db.rollback()
                print("[DEBUG] Database transaction rolled back.")
            except Exception:
                traceback.print_exc()

        finally:
            print("=" * 60)
            print("REFLECTION DEBUG: UPDATE FINISHED")
            print("=" * 60 + "\n")

    def parse_reflection_response(self, response):
        """
        Parse JSON safely, including responses wrapped in Markdown fences.
        """
        text = response.strip()

        if text.startswith("```"):
            lines = text.splitlines()

            if lines and lines[0].startswith("```"):
                lines = lines[1:]

            if lines and lines[-1].strip() == "```":
                lines = lines[:-1]

            text = "\n".join(lines).strip()

            if text.lower().startswith("json"):
                text = text[4:].strip()

        try:
            result = json.loads(text)
        except json.JSONDecodeError as exc:
            print(f"[ERROR] Invalid JSON: {exc}")
            print(f"[DEBUG] Text being parsed: {text!r}")
            return None

        return result

    def reset_counter(self, db, episode):
        try:
            episode.messages_since_last_reflection = 0
            db.commit()
            db.refresh(episode)

            print("[DEBUG] Reflection counter reset to 0.")

        except Exception:
            db.rollback()
            print("[ERROR] Failed to reset reflection counter.")
            traceback.print_exc()
            raise

    def build_prompt(
        self,
        episode,
        previous_reflections,
        conversation,
        latest_message,
        reply
    ):
        episode_text = f"""
Episode ID: {episode.episode_id}

Title: {episode.title}

Status: {episode.status}

Summary:
{episode.summary if episode.summary else "None"}
"""

        if not previous_reflections:
            reflections_text = "None"
        else:
            reflections_text = "\n".join(
                f"{item.created_at} - {item.reflection}"
                for item in previous_reflections
            )

        conversation_text = "\n".join(
            f"{message['role'].capitalize()}: {message['content']}"
            for message in conversation
        )

        return PROMPT.format(
            episode=episode_text,
            reflections=reflections_text,
            conversation=conversation_text,
            message=latest_message,
            reply=reply
        )

    def save(self, db, episode, reflection):
        new_reflection = EpisodeReflection(
            episode_id=episode.episode_id,
            reflection=reflection
        )

        db.add(new_reflection)
        db.commit()
        db.refresh(new_reflection)

        print("\n===== REFLECTION SAVED =====")
        print(f"ID: {new_reflection.id}")
        print(f"Episode: {new_reflection.episode_id}")
        print(f"Reflection: {new_reflection.reflection}")
        print(f"Created at: {new_reflection.created_at}")
        print("============================\n")
